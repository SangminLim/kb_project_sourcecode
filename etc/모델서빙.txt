# app.py

from fastapi import FastAPI
from pydantic import BaseModel

from predict import load_model, predict

app = FastAPI()

# 서버 시작 시 모델 메모리 로드
model = load_model()


# 요청 데이터 형식
class PredictRequest(BaseModel):
    text: str


# 응답 API
@app.post("/predict")
def inference(request: PredictRequest):

    result = predict(model, request.text)

    return {
        "result": result
    }

# predict.py

from transformers import pipeline

def load_model():

    model = pipeline(
        "text-classification",
        model="./model"
    )

    return model


def predict(model, text):

    result = model(text)

    return result
